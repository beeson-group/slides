% the Earth, the Moon, and the libration points L1 and L2 of the Earth-Moon system in the synodic
% frame, not to scale, with an orbit around L2 drawn as an ellipse, i.e. a loop. the points and the loop are parts
\begin{tikzpicture}
  \draw[subtle, dashed] (-0.6,0) -- (5.7,0);
  \fill[blue] (0,0) circle (0.3);
  \node[below=4pt] at (0,-0.3) {Earth};
  \fill[subtle] (4,0) circle (0.12);
  \node[below=4pt] at (4,-0.3) {Moon};
  \begin{scope}[part=points, violet, line width=0.8333pt]
    \foreach \x/\i in {3.15/1, 5/2} {
      \draw (\x-0.08,-0.08) -- (\x+0.08,0.08) (\x-0.08,0.08) -- (\x+0.08,-0.08);
      \node[above=4pt] at (\x,0.08) {$\mathcal{L}_\i$};
    }
  \end{scope}
  \begin{scope}[part=loop]
    \draw[accent, line width=0.8333pt] (5,0) ellipse (0.35 and 1.1);
  \end{scope}
\end{tikzpicture}
